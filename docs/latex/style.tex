\newcommand{\browserframe}[3]{% #1 width, #2 address, #3 image
  \begin{tikzpicture}
    \node[inner sep=0pt, outer sep=0pt] (img) {#3};
    \coordinate (tr) at ([xshift=1.4pt,yshift=16pt]img.north east);
    \coordinate (bl) at ([xshift=-1.4pt,yshift=-1.4pt]img.south west);
    \begin{pgfonlayer}{background}
      \fill[black!9, rounded corners=5pt] ([xshift=1.2pt,yshift=-2pt]bl) rectangle ([xshift=1.2pt,yshift=-2pt]tr);
      \fill[chromebg, rounded corners=5pt] (bl) rectangle (tr);
    \end{pgfonlayer}
    \draw[chromeedge, line width=0.5pt, rounded corners=5pt] (bl) rectangle (tr);
    \draw[chromeedge, line width=0.4pt] ([xshift=-0.2pt]img.north west) -- ([xshift=0.2pt]img.north east);
    \fill[dotred]    ([xshift=7pt, yshift=7.8pt]img.north west) circle (2.3pt);
    \fill[dotyellow] ([xshift=14.5pt,yshift=7.8pt]img.north west) circle (2.3pt);
    \fill[dotgreen]  ([xshift=22pt, yshift=7.8pt]img.north west) circle (2.3pt);
    \node[fill=white, draw=chromeedge, line width=0.3pt, rounded corners=3pt, inner xsep=5pt, inner ysep=1.5pt,
          minimum width=0.42*#1, font=\fontsize{6}{7}\selectfont\ttfamily, text=black!55]
      at ([yshift=7.8pt]$(img.north west)!0.5!(img.north east)$) {#2};
  \end{tikzpicture}}
